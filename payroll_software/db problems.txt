
=============== MariaDB store PROCEDURE ======================

DELIMITER //

CREATE OR REPLACE PROCEDURE `getDrInvoiceDateByDateTransactionSummary1`(
    IN `from_date` DATE,
    IN `to_date` DATE,
    IN `credit_acc_ids` JSON
)
NOT DETERMINISTIC
CONTAINS SQL
SQL SECURITY DEFINER
BEGIN
    SELECT
        dr.ExpenseDate,
        SUM(dr.Amount) AS 'Total_Amount'
    FROM dr_vouchers dr
    WHERE
        /* MariaDB optimized JSON search */
        JSON_CONTAINS(credit_acc_ids, JSON_QUOTE(CAST(dr.CreditedFromId AS CHAR)))
        AND dr.ExpenseDate BETWEEN from_date AND to_date
    GROUP BY dr.ExpenseDate
    ORDER BY dr.ExpenseDate ASC;
END //

DELIMITER ;
